# Laser defaults: what each color means, how it is drawn, and how the laser panel
# fills in that color's row the first time it sees it. Edit freely.
#
# Inkscape's Laser menu uses this file: "Set up page for laser" sets the page to the
# laser's bed and adds these colors as swatches; "Mark as Cut / Score / Engrave"
# restyles the selection.
#
# width is the outline width in mm (0.0254 mm = 0.001 in, a hairline as in Epilog).
# A role with fill = yes is drawn as a filled shape with no outline.

[Cut]
color = #ff0000
width = 0.0254
fill = no

[Score]
color = #0000ff
width = 0.0254
fill = no

[Engrave]
color = #000000
width = 0
fill = yes

# not seen in this drawing before; anything changed in the panel is remembered.

[rk-uv Cut]

frequency_khz = 30
passes = 1

[rk-uv Score]

[rk-uv Engrave]
